```python
def calculate_user_ltv():
    """
    Lifetime Value calculation based on research data
    """
    metrics = {
        'monthly_subscription': 599,  # ₹599/month based on willingness-to-pay
        'average_subscription_months': 18,  # Based on engagement sustainability
        'upsell_probability': 0.3,  # Premium features adoption
        'premium_uplift': 400,  # Additional ₹400 for premium
        'referral_value': 200,  # Word-of-mouth acquisition value
        'churn_rate': 0.05  # Monthly churn based on engagement data
    }
    
    base_ltv = (metrics['monthly_subscription'] * 
                metrics['average_subscription_months'] * 
                (1 - metrics['churn_rate']))
    
    upsell_value = (metrics['premium_uplift'] * 
                   metrics['average_subscription_months'] * 
                   metrics['upsell_probability'])
    
    total_ltv = base_ltv + upsell_value + metrics['referral_value']
    return total_ltv

# Expected LTV: ₹12,000-15,000 per user
```
