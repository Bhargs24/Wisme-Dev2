SMART CACHING & EPISODE REUSE SYSTEM

One of the most innovative aspects of WISME's data strategy is our intelligent episode caching and reuse system. This system allows us to provide high-quality content to free users while managing costs effectively.

```dart
// lib/core/content/smart_episode_cache.dart
class SmartEpisodeCache {
  static const int maxCachedEpisodes = 1000;
  static const int maxReuseCount = 50; // Prevent over-repetition
  
  final Map<String, CachedEpisode> _episodeCache = {};
  final Map<String, int> _reuseCount = {};
  final Map<String, double> _relevanceScores = {};
  
  /// Caches an episode with metadata for intelligent reuse
  Future<void> cacheEpisode({
    required String episodeId,
    required String content,
    required String category,
    required String difficulty,
    required List<String> tags,
    required String voiceId,
    required String audioUrl,
  }) async {
    final cachedEpisode = CachedEpisode(
      id: episodeId,
      content: content,
      category: category,
      difficulty: difficulty,
      tags: tags,
      voiceId: voiceId,
      audioUrl: audioUrl,
      createdAt: DateTime.now(),
      lastUsed: DateTime.now(),
      useCount: 0,
      averageRating: 0.0,
      completionRate: 0.0,
    );
    
    _episodeCache[episodeId] = cachedEpisode;
    _reuseCount[episodeId] = 0;
    
    // Maintain cache size
    if (_episodeCache.length > maxCachedEpisodes) {
      _cleanupOldEpisodes();
    }
  }
  
  /// Finds the most relevant cached episode for a user request
  Future<CachedEpisode?> findRelevantEpisode({
    required String category,
    required String difficulty,
    required List<String> userInterests,
    required String userId,
  }) async {
    final candidates = _episodeCache.values.where((episode) {
      // Basic filtering
      if (episode.category != category) return false;
      if (episode.difficulty != difficulty) return false;
      if (_reuseCount[episode.id]! >= maxReuseCount) return false;
      
      // Check if user has already heard this episode recently
      if (_hasUserHeardRecently(episode.id, userId)) return false;
      
      return true;
    }).toList();
    
    if (candidates.isEmpty) return null;
    
    // Score candidates based on relevance
    final scoredCandidates = candidates.map((episode) {
      final relevanceScore = _calculateRelevanceScore(
        episode: episode,
        userInterests: userInterests,
        userId: userId,
      );
      return _ScoredEpisode(episode: episode, score: relevanceScore);
    }).toList();
    
    // Sort by relevance score and select the best match
    scoredCandidates.sort((a, b) => b.score.compareTo(a.score));
    
    final selectedEpisode = scoredCandidates.first.episode;
    
    // Update usage statistics
    _updateEpisodeUsage(selectedEpisode.id, userId);
    
    return selectedEpisode;
  }
  
  /// Calculates relevance score for episode matching
  double _calculateRelevanceScore({
    required CachedEpisode episode,
    required List<String> userInterests,
    required String userId,
  }) {
    double score = 0.0;
    
    // Tag matching (40% weight)
    final tagMatches = episode.tags.where((tag) => 
      userInterests.contains(tag)).length;
    score += (tagMatches / episode.tags.length) * 0.4;
    
    // User engagement history (30% weight)
    final userEngagement = _getUserEngagementScore(episode.id, userId);
    score += userEngagement * 0.3;
    
    // Episode quality metrics (20% weight)
    final qualityScore = (episode.averageRating + episode.completionRate) / 2;
    score += qualityScore * 0.2;
    
    // Freshness factor (10% weight)
    final daysSinceCreation = DateTime.now().difference(episode.createdAt).inDays;
    final freshnessScore = 1.0 / (1.0 + daysSinceCreation / 30.0); // Decay over 30 days
    score += freshnessScore * 0.1;
    
    return score;
  }
  
  /// Tracks user engagement to prevent repetition
  bool _hasUserHeardRecently(String episodeId, String userId) {
    final userHistory = _getUserEpisodeHistory(userId);
    final recentEpisodes = userHistory.where((history) =>
      DateTime.now().difference(history.listenedAt).inDays < 7
    ).toList();
    
    return recentEpisodes.any((history) => history.episodeId == episodeId);
  }
  
  /// Updates episode usage statistics
  void _updateEpisodeUsage(String episodeId, String userId) {
    final episode = _episodeCache[episodeId];
    if (episode != null) {
      episode.useCount++;
      episode.lastUsed = DateTime.now();
      _reuseCount[episodeId] = (_reuseCount[episodeId] ?? 0) + 1;
      
      // Track user interaction
      _trackUserEpisodeInteraction(episodeId, userId);
    }
  }
}
```

```dart
// lib/core/audio/voice_cache_manager.dart
class VoiceCacheManager {
  // Phase 1: Predetermined voices for each category
  static const Map<String, String> categoryVoices = {
    'startup_funding': 'voice_startup_expert',
    'business_law': 'voice_legal_advisor',
    'product_design': 'voice_design_mentor',
    'tech_trends': 'voice_tech_analyst',
    'marketing': 'voice_marketing_guru',
    'finance': 'voice_financial_advisor',
    'leadership': 'voice_leadership_coach',
    'innovation': 'voice_innovation_expert',
  };
  
  // Premium voice options (additional voices for premium users)
  static const List<String> premiumVoices = [
    'voice_premium_1',
    'voice_premium_2',
    'voice_premium_3',
    'voice_premium_4',
  ];
  
  /// Gets appropriate voice for user tier and category
  String getVoiceForCategory(String category, bool isPremiumUser) {
    // Free users get predetermined category voices
    final baseVoice = categoryVoices[category] ?? 'voice_default';
    
    if (!isPremiumUser) {
      return baseVoice;
    }
    
    // Premium users can choose from additional voices
    // For now, return base voice + premium options
    return baseVoice;
  }
  
  /// Caches audio for frequently used content
  Future<void> cacheCategoryAudio(String category, String content) async {
    final voiceId = categoryVoices[category] ?? 'voice_default';
    
    // Generate audio if not already cached
    if (!await _isAudioCached(content, voiceId)) {
      final audioUrl = await _generateAudio(content, voiceId);
      await _cacheAudio(content, voiceId, audioUrl);
    }
  }
}
```

```dart
// lib/core/content/episode_generation_service.dart
class EpisodeGenerationService {
  final SmartEpisodeCache _cache;
  final VoiceCacheManager _voiceManager;
  final bool _isPremiumUser;
  
  EpisodeGenerationService({
    required SmartEpisodeCache cache,
    required VoiceCacheManager voiceManager,
    required bool isPremiumUser,
  }) : _cache = cache,
       _voiceManager = voiceManager,
       _isPremiumUser = isPremiumUser;
  
  /// Generates or retrieves an episode for the user
  Future<EpisodeData> getEpisode({
    required String category,
    required String difficulty,
    required List<String> userInterests,
    required String userId,
  }) async {
    // For free users, try to find cached episode first
    if (!_isPremiumUser) {
      final cachedEpisode = await _cache.findRelevantEpisode(
        category: category,
        difficulty: difficulty,
        userInterests: userInterests,
        userId: userId,
      );
      
      if (cachedEpisode != null) {
        return _convertToEpisodeData(cachedEpisode);
      }
    }
    
    // Generate new episode (premium users get personalized content)
    final episode = await _generateNewEpisode(
      category: category,
      difficulty: difficulty,
      userInterests: userInterests,
      isPremium: _isPremiumUser,
    );
    
    // Cache the episode for future reuse
    await _cache.cacheEpisode(
      episodeId: episode.id,
      content: episode.content,
      category: episode.category,
      difficulty: episode.difficulty,
      tags: episode.tags,
      voiceId: episode.voiceId,
      audioUrl: episode.audioUrl,
    );
    
    return episode;
  }
  
  /// Generates new episode content
  Future<EpisodeData> _generateNewEpisode({
    required String category,
    required String difficulty,
    required List<String> userInterests,
    required bool isPremium,
  }) async {
    String content;
    
    if (isPremium) {
      // Premium users get personalized content
      content = await _generatePersonalizedContent(
        category: category,
        difficulty: difficulty,
        userInterests: userInterests,
      );
    } else {
      // Free users get generic but high-quality content
      content = await _generateGenericContent(
        category: category,
        difficulty: difficulty,
      );
    }
    
    final voiceId = _voiceManager.getVoiceForCategory(category, isPremium);
    final audioUrl = await _generateAudio(content, voiceId);
    
    return EpisodeData(
      id: _generateEpisodeId(),
      title: _generateTitle(content),
      content: content,
      category: category,
      difficulty: difficulty,
      tags: _extractTags(content),
      voiceId: voiceId,
      audioUrl: audioUrl,
      duration: _calculateDuration(content),
      createdAt: DateTime.now(),
      isCached: false,
      reuseCount: 0,
      averageRating: 0.0,
      completionRate: 0.0,
    );
  }
}
```

```dart
// lib/core/analytics/analytics_processor.dart
class AnalyticsProcessor {
  /// Processes user session data for insights
  Future<void> processSessionData(LearningSession session) async {
    // Update user analytics
    await _updateUserAnalytics(session);
    
    // Update episode analytics
    await _updateEpisodeAnalytics(session);
    
    // Update cache relevance scores
    await _updateCacheRelevance(session);
    
    // Track for business insights
    await _trackBusinessMetrics(session);
  }
  
  /// Updates user engagement analytics
  Future<void> _updateUserAnalytics(LearningSession session) async {
    final userAnalytics = await _getUserAnalytics(session.userId);
    
    // Update engagement metrics
    userAnalytics.categoryEngagement[session.category] = 
      (userAnalytics.categoryEngagement[session.category] ?? 0) + 1;
    
    userAnalytics.totalListenTime += session.listenDuration ~/ 60; // Convert to minutes
    
    // Calculate completion rate
    final totalSessions = userAnalytics.totalSessions + 1;
    final completedSessions = userAnalytics.completedEpisodes.length;
    userAnalytics.completionRate = completedSessions / totalSessions;
    
    await _saveUserAnalytics(userAnalytics);
  }
  
  /// Tracks business-relevant metrics
  Future<void> _trackBusinessMetrics(LearningSession session) async {
    // Track free vs premium usage patterns
    final user = await _getUser(session.userId);
    
    if (user.tier == UserTier.FREE) {
      // Track free user engagement for ad optimization
      await _trackFreeUserEngagement(session);
    } else {
      // Track premium user value metrics
      await _trackPremiumUserValue(session);
    }
  }
}
```

```dart
class DataPrivacyManager {
  /// Anonymizes user data for analytics
  Future<Map<String, dynamic>> anonymizeUserData(UserProfile user) async {
    return {
      'tier': user.tier.toString(),
      'interests': user.interests,
      'topicProficiency': user.topicProficiency,
      'preferences': _anonymizePreferences(user.preferences),
      'totalSessions': user.totalSessions,
      'totalEpisodesCompleted': user.totalEpisodesCompleted,
      // Exclude personally identifiable information
    };
  }
  
  /// Implements data retention policies
  Future<void> enforceDataRetention() async {
    // Delete old session data (keep for 1 year)
    await _deleteOldSessions(DateTime.now().subtract(Duration(days: 365)));
    
    // Anonymize old user data (after 2 years)
    await _anonymizeOldUserData(DateTime.now().subtract(Duration(days: 730)));
    
    // Clean up unused cached episodes (after 6 months)
    await _cleanupUnusedCache(DateTime.now().subtract(Duration(days: 180)));
  }
  
  /// Provides user data export functionality
  Future<Map<String, dynamic>> exportUserData(String userId) async {
    final user = await _getUser(userId);
    final sessions = await _getUserSessions(userId);
    final analytics = await _getUserAnalytics(userId);
    
    return {
      'profile': user.toJson(),
      'sessions': sessions.map((s) => s.toJson()).toList(),
      'analytics': analytics.toJson(),
      'exported_at': DateTime.now().toIso8601String(),
    };
  }
}
```
