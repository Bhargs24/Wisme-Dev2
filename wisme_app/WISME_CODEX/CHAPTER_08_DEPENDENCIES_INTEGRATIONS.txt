```dart
// lib/core/services/supabase_integration.dart
class SupabaseIntegration {
  static const String _url = 'https://wisme-app.supabase.co';
  static const String _anonKey = 'your-anon-key';
  
  late final SupabaseClient _client;
  
  SupabaseIntegration() {
    _client = SupabaseClient(_url, _anonKey);
  }
  
  /// Initializes Supabase with proper configuration
  Future<void> initialize() async {
    // Configure real-time subscriptions
    await _configureRealtime();
    
    // Set up row-level security policies
    await _setupSecurityPolicies();
    
    // Initialize storage buckets
    await _initializeStorage();
    
    // Set up database triggers
    await _setupDatabaseTriggers();
  }
  
  /// Configures real-time subscriptions for live updates
  Future<void> _configureRealtime() async {
    // Subscribe to user data changes
    _client
      .channel('user_updates')
      .onPostgresChanges(
        event: PostgresChangeEvent.update,
        schema: 'public',
        table: 'users',
        callback: (payload) => _handleUserUpdate(payload),
      )
      .subscribe();
    
    // Subscribe to content updates
    _client
      .channel('content_updates')
      .onPostgresChanges(
        event: PostgresChangeEvent.insert,
        schema: 'public',
        table: 'episodes',
        callback: (payload) => _handleContentUpdate(payload),
      )
      .subscribe();
  }
  
  /// Sets up row-level security policies
  Future<void> _setupSecurityPolicies() async {
    // Users can only access their own data
    await _client.rpc('setup_user_policies');
    
    // Content access based on user tier
    await _client.rpc('setup_content_policies');
    
    // Analytics data access policies
    await _client.rpc('setup_analytics_policies');
  }
  
  /// Handles user data updates
  void _handleUserUpdate(PostgresChangePayload payload) {
    final userData = payload.newRecord;
    
    // Update local state
    _updateLocalUserState(userData);
    
    // Track analytics
    _trackUserUpdate(userData);
  }
  
  /// Handles content updates
  void _handleContentUpdate(PostgresChangePayload payload) {
    final contentData = payload.newRecord;
    
    // Update local content cache
    _updateContentCache(contentData);
    
    // Notify UI of new content
    _notifyContentUpdate(contentData);
  }
}
```

```dart
class CostManagement {
  /// Monitors and optimizes Supabase usage costs
  Future<void> monitorCosts() async {
    // Track database operations
    await _trackDatabaseOperations();
    
    // Monitor storage usage
    await _monitorStorageUsage();
    
    // Optimize query performance
    await _optimizeQueries();
    
    // Implement caching strategies
    await _implementCaching();
  }
  
  /// Tracks database operation costs
  Future<void> _trackDatabaseOperations() async {
    final operations = await _getDatabaseOperations();
    
    for (final operation in operations) {
      // Calculate cost based on operation type
      final cost = _calculateOperationCost(operation);
      
      // Track cost by user tier
      await _trackCostByTier(operation.userId, cost);
      
      // Alert if costs exceed thresholds
      if (cost > _getCostThreshold(operation.userTier)) {
        await _alertHighCost(operation);
      }
    }
  }
  
  /// Calculates operation cost based on type
  double _calculateOperationCost(DatabaseOperation operation) {
    switch (operation.type) {
      case OperationType.select:
        return operation.rows * 0.0001; // $0.0001 per row
      case OperationType.insert:
        return operation.rows * 0.0002; // $0.0002 per row
      case OperationType.update:
        return operation.rows * 0.0003; // $0.0003 per row
      case OperationType.delete:
        return operation.rows * 0.0001; // $0.0001 per row
    }
  }
  
  /// Gets cost threshold based on user tier
  double _getCostThreshold(UserTier tier) {
    switch (tier) {
      case UserTier.FREE:
        return 0.01; // $0.01 per day
      case UserTier.PREMIUM:
        return 0.10; // $0.10 per day
    }
  }
}

enum OperationType { select, insert, update, delete }
```

```dart
    required String voiceId,
    required UserTier tier,
  }) async {
    try {
      // Check cache first for free users
      if (tier == UserTier.FREE) {
        final cachedAudio = await _getCachedAudio(text, voiceId);
        if (cachedAudio != null) {
          return cachedAudio;
        }
      }
      
      final response = await _httpClient.post('/text-to-speech/$voiceId', data: {
        'text': text,
        'model_id': _getModelForTier(tier),
        'voice_settings': _getVoiceSettings(tier),
      });
      
      final audioUrl = await _uploadAudio(response.data, tier);
      
      // Cache audio for free users
      if (tier == UserTier.FREE) {
        await _cacheAudio(text, voiceId, audioUrl);
      }
      
      return audioUrl;
    } catch (e) {
      throw ElevenLabsException('Failed to generate audio: $e');
    }
  }
  
  /// Gets appropriate model based on user tier
  String _getModelForTier(UserTier tier) {
    switch (tier) {
      case UserTier.FREE:
        return 'eleven_monolingual_v1'; // Cheaper model
      case UserTier.PREMIUM:
        return 'eleven_multilingual_v2'; // Premium model
    }
  }
  
  /// Gets voice settings based on user tier
  Map<String, dynamic> _getVoiceSettings(UserTier tier) {
    switch (tier) {
      case UserTier.FREE:
        return {
          'stability': 0.5,
          'similarity_boost': 0.5,
        };
      case UserTier.PREMIUM:
        return {
          'stability': 0.8,
          'similarity_boost': 0.8,
          'style': 0.3,
          'use_speaker_boost': true,
        };
    }
  }
  
  /// Manages voice selection based on user tier
  String getVoiceForCategory(String category, UserTier tier) {
    // Free users get predetermined voices
    final freeVoices = {
      'startup_funding': 'voice_startup_expert',
      'business_law': 'voice_legal_advisor',
      'product_design': 'voice_design_mentor',
      'tech_trends': 'voice_tech_analyst',
    };
    
    if (tier == UserTier.FREE) {
      return freeVoices[category] ?? 'voice_default';
    }
    
    // Premium users get more voice options
    final premiumVoices = {
      ...freeVoices,
      'startup_funding': 'voice_premium_startup',
      'business_law': 'voice_premium_legal',
      'product_design': 'voice_premium_design',
      'tech_trends': 'voice_premium_tech',
    };
    
    return premiumVoices[category] ?? 'voice_premium_default';
  }

  Future<void> _cacheAudio(String text, String voiceId, String audioUrl) async {
    final cacheKey = _generateCacheKey(text, voiceId);
    await _audioCache.set(cacheKey, audioUrl, Duration(days: 30));
  }
  
  /// Retrieves cached audio for free users
  Future<String?> _getCachedAudio(String text, String voiceId) async {
    final cacheKey = _generateCacheKey(text, voiceId);
    return await _audioCache.get(cacheKey);

    final hash = sha256.convert(utf8.encode(text + voiceId));
```

```dart
// lib/core/services/service_abstraction.dart
abstract class ServiceInterface<T> {
  Future<T> get(String id);
  Future<List<T>> getAll();
  Future<T> create(T data);
  Future<T> update(String id, T data);
  Future<void> delete(String id);
}

class SupabaseService<T> implements ServiceInterface<T> {
  final SupabaseClient _client;
  final String _table;
  final T Function(Map<String, dynamic>) _fromJson;
  final Map<String, dynamic> Function(T) _toJson;
  
  SupabaseService(
    this._client,
    this._table,
    this._fromJson,
    this._toJson,
  );
  
  @override
  Future<T> get(String id) async {
    final response = await _client
      .from(_table)
      .select()
      .eq('id', id)
      .single();
    
    return _fromJson(response);
  }
  
  @override
  Future<List<T>> getAll() async {
    final response = await _client
      .from(_table)
      .select();
    
    return response.map((item) => _fromJson(item)).toList();
  }
  
  @override
  Future<T> create(T data) async {
    final response = await _client
      .from(_table)
      .insert(_toJson(data))
      .select()
      .single();
    
    return _fromJson(response);
  }
  
  @override
  Future<T> update(String id, T data) async {
    final response = await _client
      .from(_table)
      .update(_toJson(data))
      .eq('id', id)
      .select()
      .single();
    
    return _fromJson(response);
  }
  
  @override
  Future<void> delete(String id) async {
    await _client
      .from(_table)
      .delete()
      .eq('id', id);
  }
}
```

**Error Handling and Retry Logic:**

```dart
// lib/core/services/error_handler.dart
class ServiceErrorHandler {
  /// Handles service errors with retry logic
  Future<T> handleWithRetry<T>(
    Future<T> Function() operation, {
    int maxRetries = 3,
    Duration delay = const Duration(seconds: 1),
  }) async {
    int attempts = 0;
    
    while (attempts < maxRetries) {
      try {
        return await operation();
      } catch (e) {
        attempts++;
        
        if (attempts >= maxRetries) {
          throw ServiceException('Operation failed after $maxRetries attempts: $e');
        }
        
        // Wait before retrying
        await Future.delayed(delay * attempts);
      }
    }
    
    throw ServiceException('Unexpected error in retry logic');
  }
  
  /// Handles specific service errors
  Future<void> handleServiceError(dynamic error, String serviceName) async {
    if (error is NetworkException) {
      await _handleNetworkError(error, serviceName);
    } else if (error is AuthenticationException) {
      await _handleAuthError(error, serviceName);
    } else if (error is RateLimitException) {
      await _handleRateLimitError(error, serviceName);
    } else {
      await _handleGenericError(error, serviceName);
    }
  }
  
  /// Handles network errors
  Future<void> _handleNetworkError(NetworkException error, String serviceName) async {
    // Log network error
    await _logError('Network error in $serviceName: ${error.message}');
    
    // Notify user of connectivity issues
```
